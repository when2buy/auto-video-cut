```python
# src/avc/captions.py
"""Opus-style ASS subtitles + ffmpeg burn-in.

Ported from louisedesadeleer/clipify (MIT). Adapted to take Transcript objects.
"""
from __future__ import annotations
import subprocess
from pathlib import Path
from .asr import Transcript

# --- preset table (from clipify, lightly tweaked for Chinese support) ---
PRESETS: dict[str, dict] = {
    "opus":    dict(font="Arial Black", size=84, chunk=3, highlight="&H0000FFFF&", outline=8, shadow=3),
    "opus-cn": dict(font="Source Han Sans CN", size=84, chunk=4, highlight="&H0000FFFF&", outline=8, shadow=3),
    "karaoke": dict(font="Arial Black", size=92, chunk=4, highlight="&H0000FF00&", outline=6, shadow=2),
    "minimal": dict(font="Helvetica",   size=60, chunk=6, highlight=None,           outline=4, shadow=1),
}
WHITE = "&H00FFFFFF&"
OUTLINE = "&H00000000&"
PLAY_W, PLAY_H = 1080, 1920  # default 9:16 frame; caller overrides

def _fmt_time(t: float) -> str:
    h = int(t // 3600); m = int((t % 3600) // 60); s = t - h*3600 - m*60
    return f"{h}:{m:02d}:{s:05.2f}"

def build_ass(
    transcript: Transcript,
    out: Path,
    *,
    style: str = "opus",
    play_w: int = PLAY_W,
    play_h: int = PLAY_H,
) -> Path:
    P = PRESETS.get(style, PRESETS["opus"])
    if not any(s.words for s in transcript.sentences):
        raise ValueError("Transcript lacks word-level timestamps; re-transcribe with word_timestamps=True")
    words = []
    for s in transcript.sentences:
        if not s.words:
            continue
        for w in s.words:
            words.append({"start": w.start, "end": w.end, "text": w.text})
    chunks = [words[i:i + P["chunk"]] for i in range(0, len(words), P["chunk"])]
    header = f"""[Script Info]
ScriptType: v4.00+
PlayResX: {play_w}
PlayResY: {play_h}
WrapStyle: 2
ScaledBorderAndShadow: yes

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding
Style: Default,{P["font"]},{P["size"]},{WHITE},&H000000FF,{OUTLINE},&H00000000,1,0,0,0,100,100,0,0,1,{P["outline"]},{P["shadow"]},2,60,60,280,1

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text
"""
    events = []
    for chunk in chunks:
        chunk_end = chunk[-1]["end"]
        for i, w in enumerate(chunk):
            seg_start = w["start"]
            seg_end = chunk[i+1]["start"] if i+1 < len(chunk) else chunk_end
            if seg_end <= seg_start:
                seg_end = seg_start + 0.05
            if P["highlight"]:
                parts = []
                for j, ww in enumerate(chunk):
                    if j == i:
                        parts.append(f"{{\\c{P['highlight']}}}{ww['text']}{{\\c{WHITE}}}")
                    else:
                        parts.append(ww["text"])
                line = " ".join(parts)
            else:
                line = " ".join(ww["text"] for ww in chunk)
            events.append(f"Dialogue: 0,{_fmt_time(seg_start)},{_fmt_time(seg_end)},Default,,0,0,0,,{line}")
    out.write_text(header + "\n".join(events) + "\n", encoding="utf-8")
    return out

def burn_captions(
    input_video: Path,
    ass_file: Path,
    output_video: Path,
    *,
    verbose: bool = False,
) -> Path:
    """ffmpeg -vf ass=... burn the ASS file into the video."""
    cmd = [
        "ffmpeg", "-y", "-loglevel", "error",
        "-i", str(input_video),
        "-vf", f"ass={ass_file.as_posix()}",
        "-c:v", "libx264", "-preset", "fast", "-crf", "20",
        "-pix_fmt", "yuv420p",
        "-c:a", "copy",
        str(output_video),
    ]
    if verbose:
        print(f"[burn_captions] {' '.join(cmd)}")
    subprocess.run(cmd, check=True, capture_output=not verbose)
    return output_video
```

```python
# src/avc/reframe.py — top of file
"""9:16 face-tracked reframe. Uses MediaPipe Face Detection + per-frame ROI smoothing.

Single-face mode (default): crop a vertical strip centered on the dominant face,
smoothed with EMA across frames. Suitable for talks, vlogs, single-speaker.

Caveats:
- Multi-face dialog (interview style) requires speaker-tracking; out of v0.3 scope.
- Source assumed 16:9. Other aspects: scaling fallback applied.
"""
from __future__ import annotations
import json
import shutil
import subprocess
import tempfile
from dataclasses import dataclass
from pathlib import Path

@dataclass
class FaceBox:
    x: float  # normalized 0..1
    y: float
    w: float
    h: float
    score: float = 0.0

def detect_faces(image_path: Path) -> list[dict]:
    """Run MediaPipe Face Detection on a single image. Returns list of dicts with x,y,w,h,score (all 0..1)."""
    import cv2
    import mediapipe as mp
    img = cv2.imread(str(image_path))
    if img is None:
        raise FileNotFoundError(image_path)
    h, w = img.shape[:2]
    mp_fd = mp.solutions.face_detection
    with mp_fd.FaceDetection(model_selection=1, min_detection_confidence=0.5) as fd:
        rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        results = fd.process(rgb)
        out = []
        for det in (results.detections or []):
            box = det.location_data.relative_bounding_box
            out.append({
                "x": float(box.xmin), "y": float(box.ymin),
                "w": float(box.width), "h": float(box.height),
                "score": float(det.score[0]) if det.score else 0.0,
            })
        return out
```

```python
# src/avc/reframe.py — append
def track_face_over_clip(
    video: Path,
    *,
    n_samples: int = 8,
    confidence: float = 0.5,
) -> list[tuple[float, FaceBox]] | None:
    """Sample N frames evenly across the video, run face detect on each.
    Returns list of (timestamp_s, FaceBox) for frames where a face was found,
    or None if <50% of samples had a face.
    """
    info = subprocess.run(
        ["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "json", str(video)],
        check=True, capture_output=True, text=True,
    )
    dur = float(json.loads(info.stdout)["format"]["duration"])
    timestamps = [dur * (i + 0.5) / n_samples for i in range(n_samples)]
    samples: list[tuple[float, FaceBox]] = []
    with tempfile.TemporaryDirectory() as tmp:
        tmp_p = Path(tmp)
        for ts in timestamps:
            frame = tmp_p / f"f_{ts:.2f}.jpg"
            subprocess.run([
                "ffmpeg", "-y", "-loglevel", "error",
                "-ss", f"{ts:.3f}", "-i", str(video),
                "-frames:v", "1", str(frame),
            ], check=True)
            faces = detect_faces(frame)
            if not faces: continue
            best = max(faces, key=lambda f: f["score"] * f["w"] * f["h"])
            if best["score"] < confidence: continue
            samples.append((ts, FaceBox(**best)))
    if len(samples) < n_samples * 0.5:
        return None
    return samples
```

```python
# src/avc/reframe.py — append
def reframe_to_vertical(
    input_video: Path,
    output_video: Path,
    *,
    target_w: int = 1080,
    target_h: int = 1920,
    smooth_alpha: float = 0.15,  # EMA smoothing factor for face center
    fallback_center: bool = True,
    verbose: bool = False,
) -> dict:
    """Crop a 9:16 vertical strip centered on the dominant face, smoothed.

    Returns {'mode': 'face-tracked' | 'center-crop', 'samples': N, 'output': path}
    """
    src_info = subprocess.run(
        ["ffprobe", "-v", "error", "-select_streams", "v:0",
         "-show_entries", "stream=width,height", "-of", "json", str(input_video)],
        check=True, capture_output=True, text=True,
    )
    src = json.loads(src_info.stdout)["streams"][0]
    src_w, src_h = src["width"], src["height"]
    out_w_for_src = int(src_h * target_w / target_h)  # vertical strip width matched to source height

    samples = track_face_over_clip(input_video)
    if samples is None:
        if not fallback_center:
            raise RuntimeError("face tracking failed and fallback disabled")
        if verbose: print("[reframe] no face track, falling back to center crop")
        crop_x = (src_w - out_w_for_src) // 2
        x_expr = str(crop_x)
        mode = "center-crop"
        n_samples = 0
    else:
        # EMA-smooth face center x across samples
        xs = []
        smoothed = None
        for ts, face in samples:
            cx = (face.x + face.w / 2) * src_w
            if smoothed is None:
                smoothed = cx
            else:
                smoothed = smooth_alpha * cx + (1 - smooth_alpha) * smoothed
            xs.append((ts, smoothed - out_w_for_src / 2))
        # Clamp to valid range
        xs = [(ts, max(0, min(src_w - out_w_for_src, x))) for ts, x in xs]
        # Build piecewise-constant ffmpeg expression: if(lt(t,t1),x1, if(lt(t,t2),x2,...))
        expr = f"{xs[-1][1]:.0f}"
        for ts, x in reversed(xs[:-1]):
            expr = f"if(lt(t,{ts:.3f}),{x:.0f},{expr})"
        x_expr = expr
        mode = "face-tracked"
        n_samples = len(samples)

    cmd = [
        "ffmpeg", "-y", "-loglevel", "error",
        "-i", str(input_video),
        "-filter_complex",
        f"[0:v]crop={out_w_for_src}:{src_h}:x='{x_expr}':y=0,scale={target_w}:{target_h}:flags=lanczos[v]",
        "-map", "[v]", "-map", "0:a",
        "-c:v", "libx264", "-preset", "fast", "-crf", "20", "-pix_fmt", "yuv420p",
        "-c:a", "aac", "-b:a", "192k",
        str(output_video),
    ]
    if verbose: print(f"[reframe] mode={mode}, samples={n_samples}")
    subprocess.run(cmd, check=True, capture_output=not verbose)
    return {"mode": mode, "samples": n_samples, "output": str(output_video)}
```

```python
# src/avc/pipeline.py
"""Full demo pipeline: transcribe → LLM pick → cut → optional reframe → optional captions."""
from __future__ import annotations
from pathlib import Path
from typing import Any, TypedDict
import shutil
import tempfile

from .asr import transcribe
from .captions import build_ass, burn_captions
from .llm_pick import keeps_to_segments, pick_sentences
from .reframe import reframe_to_vertical
from .silence import extract_and_concat, ffprobe_duration

class PipelineResult(TypedDict):
    stages: list[str]
    in_dur: float
    out_dur: float
    ratio: float
    n_sentences: int
    n_kept: int
    n_segments: int
    artifacts: dict[str, str]

def run(
    *,
    input_video: Path,
    output_video: Path,
    target_ratio: float = 0.30,
    style_prompt: str = "保留干货、punchline、关键信息；剪掉铺垫、重复、套话",
    model: str = "gemini-2.5-pro",
    reframe: bool = False,
    captions: bool = False,
    caption_style: str = "opus",
    target_w: int = 1080,
    target_h: int = 1920,
    artifacts_dir: Path | None = None,
    verbose: bool = False,
) -> PipelineResult:
    if not input_video.exists():
        raise FileNotFoundError(input_video)
    output_video.parent.mkdir(parents=True, exist_ok=True)
    artifacts_dir = artifacts_dir or output_video.parent / f"{output_video.stem}.artifacts"
    artifacts_dir.mkdir(parents=True, exist_ok=True)

    stages = []
    artifacts: dict[str, str] = {}

    # 1. ASR (with word timestamps if captions requested)
    if verbose: print(f"[pipeline] 1/5 transcribe (word_timestamps={captions})")
    transcript = transcribe(input_video, word_timestamps=captions, verbose=verbose)
    stages.append("asr")
    transcript_json = artifacts_dir / "transcript.json"
    transcript_json.write_text(transcript.to_json())
    artifacts["transcript"] = str(transcript_json)

    # 2. LLM pick
    if verbose: print(f"[pipeline] 2/5 llm_pick (target_ratio={target_ratio})")
    keep = pick_sentences(transcript, target_ratio=target_ratio, style_prompt=style_prompt, model=model, verbose=verbose)
    stages.append("llm_pick")
    artifacts["keep_indices"] = str(keep)

    # 3. Cut
    if verbose: print(f"[pipeline] 3/5 cut ({len(keep)} sentences)")
    segments = keeps_to_segments(transcript, keep)
    cut_video = artifacts_dir / "cut.mp4"
    extract_and_concat(input_video, cut_video, segments, verbose=verbose)
    stages.append("cut")
    artifacts["cut"] = str(cut_video)

    current = cut_video

    # 4. Optional reframe
    if reframe:
        if verbose: print(f"[pipeline] 4/5 reframe to {target_w}x{target_h}")
        reframed = artifacts_dir / "reframed.mp4"
        reframe_to_vertical(current, reframed, target_w=target_w, target_h=target_h, verbose=verbose)
        stages.append("reframe")
        artifacts["reframed"] = str(reframed)
        current = reframed
    else:
        if verbose: print(f"[pipeline] 4/5 reframe SKIPPED (reframe=False)")

    # 5. Optional captions: re-transcribe the cut video for synced word timestamps,
    #    then build ASS, then burn.
    if captions:
        if verbose: print(f"[pipeline] 5/5 captions (re-transcribe cut, build ASS, burn)")
        cut_transcript = transcribe(current, word_timestamps=True, verbose=verbose)
        ass = artifacts_dir / "captions.ass"
        # Match ASS PlayRes to current frame size
        if reframe:
            build_ass(cut_transcript, ass, style=caption_style, play_w=target_w, play_h=target_h)
        else:
            # Use source aspect
            import json, subprocess
            info = json.loads(subprocess.run(
                ["ffprobe", "-v", "error", "-select_streams", "v:0",
                 "-show_entries", "stream=width,height", "-of", "json", str(current)],
                check=True, capture_output=True, text=True,
            ).stdout)["streams"][0]
            build_ass(cut_transcript, ass, style=caption_style, play_w=info["width"], play_h=info["height"])
        artifacts["captions_ass"] = str(ass)
        burn_captions(current, ass, output_video, verbose=verbose)
        stages.append("captions")
    else:
        if verbose: print(f"[pipeline] 5/5 captions SKIPPED — copying current to output")
        shutil.copy2(current, output_video)

    return {
        "stages": stages,
        "in_dur": ffprobe_duration(input_video),
        "out_dur": ffprobe_duration(output_video),
        "ratio": ffprobe_duration(output_video) / ffprobe_duration(input_video),
        "n_sentences": len(transcript.sentences),
        "n_kept": len(keep),
        "n_segments": len(segments),
        "artifacts": artifacts,
    }
```
